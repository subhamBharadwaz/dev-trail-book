## The idea

Accessible interfaces give controls meaningful names, expose their state, preserve visible focus, and follow predictable keyboard behavior.

## Why it matters

Use native elements for simple interactions and a tested component library for composite widgets. ui-sdk handles behavior while your styles define the appearance.

## A practical example

```tsx
import { Tabs, TabsList, TabsTab, TabsPanel } from "@ui-sdk/react/tabs";

export function Lesson() {
  return (
    <Tabs defaultValue="notes">
      <TabsList aria-label="Lesson sections">
        <TabsTab value="notes">Notes</TabsTab>
        <TabsTab value="code">Code</TabsTab>
      </TabsList>
      <TabsPanel value="notes">Read the explanation.</TabsPanel>
      <TabsPanel value="code">Explore the example.</TabsPanel>
    </Tabs>
  );
}
```
